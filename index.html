import sys import time import requests def banner(): print("=" * 50) print(" DISCORD SIMULATED PAYLOAD DISPATCHER") print("=" * 50) def send_spam(webhook_url, target_user, count): payloads = [ f"Hey <@{target_user}>, check out this free nitro link: https://discord-nitro-gift.com/free?user={target_user}", f"Attention <@{target_user}>: Your account was flagged for suspicious marketplace activity.", f"<@{target_user}> You won the community raffle! Claim your prize here immediately.", f"URGENT <@{target_user}>: Verify your email to avoid automatic account termination." ] headers = {'Content-Type': 'application/json'} print(f"\n[+] Targeting user identifier: {target_user}") print(f"[+] Dispatching {count} payloads...\n") for i in range(1, count + 1): msg = payloads[i % len(payloads)] data = {"content": msg} try: response = requests.post(webhook_url, json=data, headers=headers) if response.status_code == 204 or response.status_code == 200: print(f"[>] Payload {i}/{count} delivered successfully.") else: print(f"[!] Rate-limited or error: {response.status_code}") except Exception as e: print(f"[!] Connection error: {e}") time.sleep(0.5) if __name__ == "__main__": banner() hook = input("Enter Discord Webhook URL: ").strip() user = input("Enter target username or ID: ").strip() try: num = int(input("Enter number of messages to send: ").strip()) except ValueError: num = 5 if hook and user: send_spam(hook, user, num) else: print("[!] Missing required inputs.")
